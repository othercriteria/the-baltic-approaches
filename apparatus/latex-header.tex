% (needspace.sty is not in texliveSmall; this is the package's core
% macro inlined — a stretchable penalty that takes the page break
% early when less than #1 remains.)
\makeatletter
\providecommand{\needspace}[1]{%
  \begingroup
    \setlength{\dimen@}{#1}%
    \vskip\z@\@plus\dimen@
    \penalty -100\vskip\z@\@plus-\dimen@
  \endgroup}
\makeatother

\clubpenalty=10000
\brokenpenalty=10000

\emergencystretch=1.5em

% Declare the intended spread layout in the PDF catalog: odd
% pages (rectos) on the right, first page alone. Conforming
% viewers honor it; Chrome's viewer ignores it and mis-pairs
% spreads regardless (known Chrome limitation, 2026-07-27).
% Trade build only — the screen variant is one-sided. Passed as
% a load-time option: pandoc's template loads hyperref AFTER

\makeatletter
\if@twoside\PassOptionsToPackage{pdfpagelayout=TwoPageRight}{hyperref}\fi
\makeatother

% Blank versos inserted by openright must be truly blank
% (emptypage.sty is not in texliveSmall; standard redefinition).
\makeatletter
\def\cleardoublepage{\clearpage\if@twoside\ifodd\c@page\else
  \hbox{}\thispagestyle{empty}\newpage
  \if@twocolumn\hbox{}\newpage\fi\fi\fi}
\makeatother

% Scene-break asterism, applied by apparatus/scenebreak.lua. The
% \needspace reserves 8 baselines, not 4 — with fewer, a following
% keep-together block can re-break the page and strand the asterism
% at the foot (WB's bug, five folios).
\newcommand{\scenebreak}{\par\needspace{8\baselineskip}%
  \bigskip{\centering*\enspace*\enspace*\par}\nobreak\bigskip\nobreak}

% --- Chapter head: two-deck (design review fork 3). Small
% --- letterspaced caps "CHAPTER N" over the title; starred form
% --- (Contents) is the title deck alone.
\makeatletter
\def\@makechapterhead#1{%
  \vspace*{30\p@}%
  {\parindent \z@ \raggedright
    \ifnum \c@secnumdepth >\m@ne
      {\footnotesize\addfontfeature{LetterSpace=15}%
        \MakeUppercase{\@chapapp\ \thechapter}\par}%
      \vskip 12\p@
    \fi
    {\huge\bfseries #1\par}%
    \nobreak
    \vskip 40\p@
  }}
\def\@makeschapterhead#1{%
  \vspace*{30\p@}%
  {\parindent \z@ \raggedright
    {\huge\bfseries #1\par}%
    \nobreak
    \vskip 40\p@
  }}
\makeatother

% --- Contents: austere one-page setting (number, title, folio —
% --- no leaders, normal weight, tight leading; the default
% --- book-class 1em-per-entry air spills 22 chapters onto a
% --- second page and breaks the plate facing-spread).
\makeatletter
\renewcommand*\l@chapter[2]{%
  \ifnum \c@tocdepth >\m@ne
    \addpenalty{-\@highpenalty}%
    \vskip 0.35em \@plus\p@
    \setlength\@tempdima{1.8em}%
    \begingroup
      \parindent \z@ \rightskip \@pnumwidth
      \parfillskip -\@pnumwidth
      \leavevmode
      \advance\leftskip\@tempdima
      \hskip -\leftskip
      #1\nobreak\hfil \nobreak\hb@xt@\@pnumwidth{\hss #2}\par
      \penalty\@highpenalty
    \endgroup
  \fi}
\makeatother

% --- Running heads (design review fork 4): letterspaced small
% --- caps, no numerals, folio at the outer margin; chapter
% --- openings fall to plain (folio centered in the foot); the
% --- front matter runs empty via \StartFront below.
\usepackage{fancyhdr}
\setlength{\headheight}{14pt}
\newcommand{\rhfont}{\footnotesize\scshape\addfontfeature{LetterSpace=8}}
\renewcommand{\chaptermark}[1]{\markboth{#1}{}}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyhead[LE]{\thepage\hspace{1.5em}{\rhfont\MakeLowercase{The Baltic Approaches}}}
\fancyhead[RO]{{\rhfont\MakeLowercase{\leftmark}}\hspace{1.5em}\thepage}

% --- Front/main matter switches (avoids the pandoc-template
% --- \frontmatter folio-reset gotcha entirely). \StartFront: no
% --- heads, no folios anywhere in the front (plain is aliased to
% --- empty so \chapter* pages comply). \StartMain: restore plain,
% --- folio 1 arabic on a recto, heads on.
\makeatletter
\let\ps@plainsaved\ps@plain
\newcommand{\StartFront}{\let\ps@plain\ps@empty\pagestyle{empty}}
\newcommand{\StartMain}{\let\ps@plain\ps@plainsaved
  \cleardoublepage\pagenumbering{arabic}\pagestyle{fancy}}
\makeatother

% --- Set-off message/teleprinter blocks (design review punch list
% --- item 4), applied via ::: message divs in the drafts +
% --- apparatus/chapters.lua. Smaller, faintly letterspaced, ragged
% --- right, never hyphenated: a different paper stock. Inline caps
% --- runs in ordinary prose stay untreated by design.
\newenvironment{messageblock}{%
  \par\medskip\small
  \addfontfeature{LetterSpace=3}%
  \raggedright
  \hyphenpenalty=10000\exhyphenpenalty=10000
  \setlength{\parindent}{0pt}%
}{\par\medskip}
