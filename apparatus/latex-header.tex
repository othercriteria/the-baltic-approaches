% --- Composition (borrowed from white-buffalo's preamble, tuned
% --- against the identical 5.5x8.5in / Pagella / 11pt page; see
% --- planning/assembly.md for provenance and the gotcha list).
% (needspace.sty is not in texliveSmall; this is the package's core
% macro inlined — a stretchable penalty that takes the page break
% early when less than #1 remains.)
\makeatletter
\providecommand{\needspace}[1]{%
  \begingroup
    \setlength{\dimen@}{#1}%
    \vskip\z@\@plus\dimen@
    \penalty -100\vskip\z@\@plus-\dimen@
  \endgroup}
\makeatother
\frenchspacing
\raggedbottom
\widowpenalty=10000
\clubpenalty=10000
\brokenpenalty=10000
\finalhyphendemerits=10000000
\emergencystretch=1.5em

% Scene-break asterism, applied by apparatus/scenebreak.lua. The
% \needspace reserves 8 baselines, not 4 — with fewer, a following
% keep-together block can re-break the page and strand the asterism
% at the foot (WB's bug, five folios).
\newcommand{\scenebreak}{\par\needspace{8\baselineskip}%
  \bigskip{\centering*\enspace*\enspace*\par}\nobreak\bigskip\nobreak}
